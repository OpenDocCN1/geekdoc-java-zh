        this.length = length;
        this.width = width;
    }
    public double area() {

    }
    public double perimeter() {

    }
}

```java
public static void bubbleSort(int[] arr) {
    int n = arr.length;
    for (int i = 0; i < n-1; i++)
        for (int j = 0; j < n-i-1; j++)
            if (arr[j] > arr[j+1]) {
                int temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
            }
}
```

public static int factorial(int n) {
    if (n == 0) return 1;
    return n * factorial(n - 1);
}
