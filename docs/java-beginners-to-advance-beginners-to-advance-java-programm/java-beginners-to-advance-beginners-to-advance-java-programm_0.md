public class MyPagination {

        this.list = list;

        this.pagesize = pagesize;
        this.maxPage = getMaxPage();
        try {

                    if (i >= recordCount) break;
                } catch (Exception e) {}
                newList.add(list.get(i));
            }
        } catch (Exception e) {

        }

    }
