        int i = low - 1;

                int temp = arr[i];

                arr[i] = arr[j];

                arr[j] = temp;

            }

        }

        int temp = arr[i + 1];

        arr[i + 1] = arr[high];

        arr[high] = temp;

        return i + 1;

    }

}

        }

        while (i < n1 && j < n2) {

            if (leftArray[i] <= rightArray[j]) {

                arr[k] = leftArray[i];

                i++;

            } else {

                arr[k] = rightArray[j];

            }

        }

            arr[k] = leftArray[i];

            i++;

            k++;

        }

        while (j < n2) {

            arr[k] = rightArray[j];

            j++;

            k++;

        }

    }

                left = mid + 1;

        }

        return -1;

    }
