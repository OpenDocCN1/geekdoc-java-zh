```go
package main

import (
	"html/template"
	"net/http"
)

func formHandler(w http.ResponseWriter, r *http.Request) {
	tmpl, err := template.ParseFiles("index.html")
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	tmpl.Execute(w, nil)
}

func submitHandler(w http.ResponseWriter, r *http.Request) {
	username := r.FormValue("username")
	password := r.FormValue("password")

	// 处理表单数据
	tmpl, err := template.New("result").Parse("Username: {{.Username}}<br>Password: {{.Password}}")
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	data := struct {
		Username string
		Password string
	}{Username: username, Password: password}

	tmpl.Execute(w, data)
}

func main() {
	http.HandleFunc("/", formHandler)
	http.HandleFunc("/submit", submitHandler)
	http.ListenAndServe(":8080", nil)
}
```

## 第 14 章：Go 中的数据库交互

### 操作 SQL 数据库

SQL（结构化查询语言）数据库是应用程序中最常用的数据库类型之一。

#### 连接 SQL 数据库

要连接 SQL 数据库，你需要相应的驱动程序。例如，`database/sql`包提供了通用接口。

```go
package main

import (
	"database/sql"
	"fmt"
	_ "github.com/go-sql-driver/mysql" // 驱动
)

func main() {
	// 打开数据库
	db, err := sql.Open("mysql", "user:password@tcp(127.0.0.1:3306)/dbname")
	if err != nil {
		panic(err)
	}
	defer db.Close()

	// 检查连接是否可用
	err = db.Ping()
	if err != nil {
		panic(err)
	}
	fmt.Println("成功连接到数据库")
}
```

#### 执行 SQL 查询

你可以使用`Exec`方法执行 SQL 语句。

```go
func main() {
	// ... 之前的连接代码 ...
	_, err := db.Exec("INSERT INTO users (name, age) VALUES (?, ?)", "Alice", 25)
	if err != nil {
		fmt.Println("执行失败:", err)
		return
	}
	fmt.Println("插入成功")
}
```

#### 查询数据

你可以使用`Query`方法从数据库中检索数据。

```go
func main() {
	// ... 之前的连接代码 ...
	rows, err := db.Query("SELECT id, name, age FROM users")
	if err != nil {
		fmt.Println("查询失败:", err)
		return
	}
	defer rows.Close()

	for rows.Next() {
		var id int
		var name string
		var age int
		if err := rows.Scan(&id, &name, &age); err != nil {
			fmt.Println("扫描失败:", err)
			continue
		}
		fmt.Printf("ID: %d, Name: %s, Age: %d\n", id, name, age)
	}
}
```
